function on_activate(ctx)
  ctx.subscribe("document:changed", function()
    local text = ctx.get_document_text()
    local count = 0
    for _ in text:gmatch("%a+") do count = count + 1 end
    ctx.set_view_content("wc", "<p>Слов: " .. count .. "</p>")
  end)
end
